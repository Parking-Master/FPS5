/*
 * Rules:
  * 1. A bot's destination will always be an Access Point. It will never stop on an Entry Point or a Navigational Point.
  * 2. Since the system doesn't have collision detection for walls, it relies on Entry Points to tell the bot where it
  * can go in order to navigate to an Access Point without going through walls.
  * 3. If a required Entry Point is blocked by a wall from the current bot position, the script will refer the function
  * to an array of Navigational Points to go to first. Once the bot has gotten to the last Navigational Point in the array,
  * it will then go directly to the Entry Point, then the Access Point.

 * Definitions:
  * Access Point: A spot on the map that players would usually check for enemies, where the bot will also mainly go to.
  * Entry Point: A spot in between multiple Access Points that allow the bot to get to an Access Point from their current position.
  * Navigational Point: One or more points that allow the bot to get to an Entry Point that is blocked by an obstruction (such as a wall or ramp).
*/

access_points = [
{
  point: [-3,0,-16],
  id: 0,
  instructions: {
    1: [entry_points[0]],
    2: [entry_points[1],access_points[1].point,entry_points[0]],
    3: [access_points[2].point,entry_points[1],access_points[1].point,entry_points[0]]
  }
},
{
  point: [13.5,0,-11],
  id: 1,
  instructions: {
    0: [entry_points[0]],
    2: [entry_points[1]],
    3: [access_points[2].point,entry_points[1]]
  }
},
{
  point: [11.5,0,10.5],
  id: 2,
  instructions: {
    0: [entry_points[0],access_points[1].point,entry_points[1]],
    1: [entry_points[1]],
    3: [access_points[3].point]
  }
},
{
  point: [0,0,0],
  id: 3,
  instructions: {
    0: [entry_points[0],access_points[1].point,entry_points[1],access_points[2].point],
    1: [entry_points[1],access_points[2].point],
    2: [access_points[2].point]
  }
}
];
entry_points=[[6,0,-17],[18.2,0,5.5]];
function getInstructions(current_access_point, target_access_point) {
  const instructions = access_points[target_access_point].instructions[current_access_point];
  return instructions;
}
target = access_points[1];
instructions = getInstructions(3, target.id);
for (let i = 0; i < instructions.length + 1; i++) {
  let point = instructions[i];
  setTimeout(() => {
    if (typeof point == "undefined") point = target.point;
    point[1] = 1;
    const interval = setInterval(() => {
      const v = new THREE.Vector3(point[0], point[1], point[2]);
      if (playerCollider.start.distanceTo(v) < 1) return clearInterval(interval);
      playerCollider.end.lerp(v, 0.25);
      playerCollider.start.lerp(v, 0.25);
    }, 10);
  }, i * 4000);
}
